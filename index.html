เลือกโหมดข้อมูล และตั้งค่าสคริปต์สำหรับบันทึกข้อมูลเข้า Google Sheets

โค้ด Google Apps Script (รองรับอัปโหลดรูปภาพ) 
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  
  sheet.appendRow([
    data.petName,
    data.petType,
    data.breed,
    data.ownerName,
    data.ownerPhone,
    data.area,
    data.vaccineName,
    data.lastDate,
    data.nextDate,
    data.photoUrl || ''
  ]);
  
  return ContentService.createTextOutput(JSON.stringify({result: "success"}))
    .setMimeType(ContentService.MimeType.JSON);
}

โครงสร้างลำดับคอลัมน์ใน Sheet (เรียงตาม A-J):

1.ชื่อสัตว์ | 2.ประเภท | 3.สายพันธุ์ | 4.ชื่อเจ้าของ | 5.เบอร์โทร | 6.หมู่บ้าน | 7.ชื่อวัคซีน | 8.วันที่ฉีดล่าสุด | 9.วันที่นัดถัดไป | 10.รูปถ่าย(URL)
