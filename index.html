การเชื่อมต่อ Google Sheets API

สำหรับเจ้าหน้าที่ไอที อบต.ห้วยยาบ

ระบบรองรับการเชื่อมต่อกับ Google Sheets ผ่าน **Google Apps Script Web App** เพื่อจัดเก็บข้อมูลแบบ Real-time บนคลาวด์ของหน่วยงาน

// ตัวอย่าง Google Apps Script (doPost)
function doPost(e) {
  var sheet = SpreadsheetApp.getActiveSpreadsheet().getActiveSheet();
  var data = JSON.parse(e.postData.contents);
  sheet.appendRow([new Date(), data.id, data.name, data.species, data.owner, data.phone, data.village, data.lastVaccine, data.imageUrl]);
  return ContentService.createTextOutput(JSON.stringify({result: "success"})).setMimeType(ContentService.MimeType.JSON);
}
